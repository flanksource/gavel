import (
	"encoding/base64"
	"encoding/json"
	"fmt"
	"os"

	"github.com/onsi/ginkgo/v2"
	"github.com/onsi/ginkgo/v2/types"
)

var _ = ginkgo.ReportAfterSuite("gavel structured metadata", func(report types.Report) {
	ginkgo.PauseOutputInterception()
	defer ginkgo.ResumeOutputInterception()

	payload, err := json.Marshal([]types.Report{report})
	if err != nil {
		panic(fmt.Errorf("marshal Ginkgo report: %w", err))
	}
	encoded := base64.StdEncoding.EncodeToString(payload)
	const chunkSize = 512
	count := (len(encoded) + chunkSize - 1) / chunkSize
	// Succinct Ginkgo output leaves the cursor mid-line ("•"); frames must start a line.
	if _, err := fmt.Fprintln(os.Stdout); err != nil {
		panic(fmt.Errorf("write Ginkgo report: %w", err))
	}
	for i := 0; i < count; i++ {
		end := (i + 1) * chunkSize
		if end > len(encoded) {
			end = len(encoded)
		}
		if _, err := fmt.Fprintf(os.Stdout, "GAVEL_GINKGO_V1:%d/%d:%s\n", i, count, encoded[i*chunkSize:end]); err != nil {
			panic(fmt.Errorf("write Ginkgo report: %w", err))
		}
	}
})
