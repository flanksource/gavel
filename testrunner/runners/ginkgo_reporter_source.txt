// Aliased imports: this file joins the user's package, so an unaliased or dot
// import would redeclare any package-level name it shares (e.g. Labels, types).
import (
	gavelReporterBase64 "encoding/base64"
	gavelReporterJSON "encoding/json"
	gavelReporterFmt "fmt"
	gavelReporterOS "os"

	gavelReporterGinkgo "github.com/onsi/ginkgo/v2"
	gavelReporterTypes "github.com/onsi/ginkgo/v2/types"
)

var _ = gavelReporterGinkgo.ReportAfterSuite("gavel structured metadata", func(report gavelReporterTypes.Report) {
	gavelReporterGinkgo.PauseOutputInterception()
	defer gavelReporterGinkgo.ResumeOutputInterception()

	payload, err := gavelReporterJSON.Marshal([]gavelReporterTypes.Report{report})
	if err != nil {
		panic(gavelReporterFmt.Errorf("marshal Ginkgo report: %w", err))
	}
	encoded := gavelReporterBase64.StdEncoding.EncodeToString(payload)
	const chunkSize = 512
	count := (len(encoded) + chunkSize - 1) / chunkSize
	// Succinct Ginkgo output leaves the cursor mid-line ("•"); frames must start a line.
	if _, err := gavelReporterFmt.Fprintln(gavelReporterOS.Stdout); err != nil {
		panic(gavelReporterFmt.Errorf("write Ginkgo report: %w", err))
	}
	for i := 0; i < count; i++ {
		end := (i + 1) * chunkSize
		if end > len(encoded) {
			end = len(encoded)
		}
		if _, err := gavelReporterFmt.Fprintf(gavelReporterOS.Stdout, "GAVEL_GINKGO_V1:%d/%d:%s\n", i, count, encoded[i*chunkSize:end]); err != nil {
			panic(gavelReporterFmt.Errorf("write Ginkgo report: %w", err))
		}
	}
})
